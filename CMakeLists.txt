cmake_minimum_required(VERSION 3.18)
project(CliPP2Backends LANGUAGES CXX)

option(CLIPP2_BUILD_CUDA "Build the optional CUDA backend" OFF)
option(CLIPP2_OPENMP "Parallelize independent CPU cells and refit tasks" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Same sorted relative-name/SHA256 manifest as kernel/build_identity.py, with no
# Python, PyTorch, CUDA toolkit or NVIDIA driver needed for a CPU-only build.
function(clipp2_backend kind)
    file(GLOB headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/kernel/csrc/common/*.h")
    file(GLOB sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/kernel/csrc/${kind}/*")
    set(inputs ${headers} ${sources})
    list(SORT inputs)
    # Recompute embedded identities after content edits, not only glob changes.
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${inputs})
    set(manifest "")
    foreach(path IN LISTS inputs)
        file(SHA256 "${path}" digest)
        file(RELATIVE_PATH name "${CMAKE_CURRENT_SOURCE_DIR}/kernel/csrc" "${path}")
        string(APPEND manifest "${name}:${digest}\n")
    endforeach()
    string(SHA256 identity "${manifest}")
    add_library(clipp2_native_${kind} SHARED ${sources})
    set_target_properties(clipp2_native_${kind} PROPERTIES PREFIX "" OUTPUT_NAME "_native_${kind}")
    target_compile_definitions(clipp2_native_${kind} PRIVATE CLIPP2_BACKEND_BUILD_ID="${identity}")
endfunction()

clipp2_backend(cpu)
target_compile_options(clipp2_native_cpu PRIVATE -O3 -fno-fast-math -ffp-contract=off)
if(CLIPP2_OPENMP)
    find_package(OpenMP REQUIRED)
    target_link_libraries(clipp2_native_cpu PRIVATE OpenMP::OpenMP_CXX)
endif()

if(CLIPP2_BUILD_CUDA)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES 80 89 90)
    endif()
    enable_language(CUDA)
    clipp2_backend(cuda)
    set_target_properties(clipp2_native_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(clipp2_native_cuda PRIVATE
        $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
endif()
